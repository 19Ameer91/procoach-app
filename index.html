اسحب اللاعبين لتغيير الخطة التكتيكية

// إنشاء لاعبين قابلين للسحب والإفلات function setupPlayers() { const board = document.getElementById('tactics-board'); for(let i=1; i<=11; i++) { const dot = document.createElement('div'); dot.className = 'player-node'; dot.innerText = i; dot.style.top = (Math.random() * 80) + "%"; dot.style.left = (Math.random() * 80) + "%"; // إضافة خاصية السحب let isDragging = false; dot.onmousedown = (e) => isDragging = true; window.onmousemove = (e) => { if(isDragging) { const rect = board.getBoundingClientRect(); dot.style.left = (e.clientX - rect.left - 15) + "px"; dot.style.top = (e.clientY - rect.top - 15) + "px"; } }; window.onmouseup = () => isDragging = false; board.appendChild(dot); } } function saveTactics() { alert("تم حفظ الخطة التكتيكية في السحابة بنجاح يا قائد!"); }
